\section{Proofs}
\label{app:theory}

\subsection{Proof of Mini-Batch Exposure}
\label{app:batch-proof}

\begin{proof}[Proof of Proposition~\ref{prop:batch-exposure}]
For a fixed anchor $i$ and relation $j\neq i$, a uniformly shuffled batch of size $B$ includes $j$ with probability $(B-1)/(N-1)$. Independence across reshuffles therefore gives
\begin{equation}
\Pr[j\notin\mathcal O_i^{(T)}]
=\left(1-\frac{B-1}{N-1}\right)^T.
\end{equation}
Weighting this event by $p_i^T(j)$ and summing over $j$ proves Eq.~\eqref{eq:batch-exposure}, because $\sum_jp_i^T(j)=1$.
\end{proof}

\subsection{Proof of the Local-to-Global Bound}
\label{app:local-global-proof}

\begin{proof}[Proof of Proposition~\ref{prop:global-geometry}]
Splitting the distortion for anchor $i$ over selected and unselected relations gives
\begin{align}
\sum_{j\neq i}p_i^T(j)\ell_{ij}
&=p_i^T(C_i)\epsilon_i
+\sum_{j\notin C_i}p_i^T(j)\ell_{ij}\\
&\leq \epsilon_i+\delta_i,
\end{align}
because $p_i^T(C_i)\leq1$ and $0\leq\ell_{ij}\leq1$. Averaging over anchors proves $\mathcal E\leq\bar\epsilon+\bar\delta$.
\end{proof}

\subsection{Proof of the Teacher-Proportional Missed-Mass Bound}
\label{app:proportional-bound}

Let $C_i^{(T)}$ be the union of relations selected for anchor $i$ through epoch $T$, and let $\bar\delta_T=N^{-1}\sum_i[1-p_i^T(C_i^{(T)})]$. Suppose RIPPLE makes at least $m$ teacher-proportional draws per anchor and epoch. A relation retained deterministically has zero miss probability. For any remaining relation $j$, proportional sampling without replacement selects $j$ at each draw with conditional probability at least $p_i^T(j)$, because removing other entries can only decrease the remaining normalizer. Its probability of being missed by at least $mT$ draws is therefore at most $(1-p_i^T(j))^{mT}$. Consequently,
\begin{equation}
\mathbb E[\bar\delta_T]
\leq\frac1N\sum_i\sum_{j\neq i}
p_i^T(j)(1-p_i^T(j))^{mT}.
\label{eq:proportional-missed-mass}
\end{equation}
Deterministic high-mass retention and additional candidates can only reduce missed mass.

Combining Eq.~\eqref{eq:proportional-missed-mass} with Proposition~\ref{prop:global-geometry} gives a finite-epoch upper bound: global distortion is controlled by selected-support error plus the residual teacher mass implied by the sampling process.

\subsection{Proof of the Concentrated-Support Comparison}
\label{app:concentrated-support}

When $B$ is small relative to $N$, Proposition~\ref{prop:batch-exposure} requires $\Theta(N\log(1/\varepsilon))$ batch relation opportunities per anchor to reduce expected missed mass below $\varepsilon$. If $p_i^T$ is uniform on $h$ relevant relations, Eq.~\eqref{eq:proportional-missed-mass} reduces to
\begin{equation}
\mathbb E[\delta_{i,T}]
\leq (1-1/h)^{mT},
\end{equation}
which requires $\Theta(h\log(1/\varepsilon))$ teacher-proportional draws. Thus matched relational compute gives a factor-$\Theta(N/h)$ separation in the concentrated-support regime. This comparison is not distribution-free: real teacher targets can be heavy-tailed, in which case the empirical retained-mass curve determines the advantage.
