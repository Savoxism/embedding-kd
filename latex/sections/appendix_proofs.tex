\section{Proofs}
\label{app:proofs}

Throughout, $\lVert v\rVert_1=\sum_j|v_j|$ and
$\mathrm{TV}(p,q)=\tfrac12\lVert p-q\rVert_1$. For a distribution $p$
and a set $C$ with $p(C)>0$, $p|_C$ denotes the conditional distribution
on $C$, extended by zero outside $C$ when two distributions are compared
on a common domain.

\subsection{Proof of Proposition~\ref{prop:coverage}}
\label{app:proof-coverage}

The batch containing anchor $i$ includes $B-1$ of the remaining $N-1$
examples uniformly without replacement. Thus, for any fixed $j\neq i$,
\begin{equation}
\Pr[j\in\mathcal L_{\mathcal B_e}(i)]=\frac{B-1}{N-1}.
\end{equation}
Under independent reshuffling, the probability that $j$ has appeared with
$i$ after $T$ epochs is
\begin{equation}
q_T=1-\left(1-\frac{B-1}{N-1}\right)^T.
\end{equation}
Linearity of expectation over the $k$ members of $\mathcal N_k^T(i)$ gives
Eq.~\eqref{eq:coverage-exact}; the upper bound follows from
$1-(1-x)^T\leq Tx$ for $x\in[0,1]$.

Every realized row-$i$ term in Eq.~\eqref{eq:batchrel} evaluates student
scores only at columns in $\mathcal L_{\mathcal B_e}(i)$. Across epochs, the
objective therefore depends on row $i$ only through
$\{c^S_{ij}:j\in\mathcal O_i^{(T)}\}$. Any two score arrays that agree on
this set have the same realized batch-relational objective, irrespective of
their values outside it. This proves the support-level under-identification.
\qed

\subsection{Proof of Theorem~\ref{thm:profile-geometry}}
\label{app:proof-profile}

Fix $r$ and let $a_i=\mathrm{TV}(g^T_{i,r},g^S_{i,r})$. The reverse triangle
inequality gives
\begin{equation}
\label{eq:reverse-tv}
\left|
\mathrm{TV}(g^T_{i,r},g^T_{j,r})
-\mathrm{TV}(g^S_{i,r},g^S_{j,r})
\right|
\leq a_i+a_j.
\end{equation}
By Pinsker's inequality,
$a_i\leq\sqrt{D_{\mathrm{KL}}(g^T_{i,r}\|g^S_{i,r})/2}$.
Under the uniform assumption, Eq.~\eqref{eq:reverse-tv} is therefore at most
$2\sqrt{\varepsilon_r/2}=\sqrt{2\varepsilon_r}$.

For the average statement, Jensen's inequality yields
\begin{equation}
\frac1N\sum_i a_i
\leq
\sqrt{\frac{1}{2N}\sum_i
D_{\mathrm{KL}}(g^T_{i,r}\|g^S_{i,r})}
=\sqrt{\frac{\bar\varepsilon_r}{2}}.
\end{equation}
Averaging Eq.~\eqref{eq:reverse-tv} over ordered pairs $(i,j)$ gives
$2N^{-1}\sum_i a_i\leq\sqrt{2\bar\varepsilon_r}$, completing the proof.
\qed

\subsection{Proofs for Sampled Supervision}
\label{app:proof-sampling}

\paragraph{Proof of Lemma~\ref{lem:sampling}.}
An atom $j$ is absent from $m$ independent draws from $p$ with probability
$(1-p_j)^m$. Hence
\begin{equation}
\mathbb E[p(C_m^c)]=\sum_jp_j(1-p_j)^m.
\end{equation}
For any $t>0$, atoms with $p_j\geq t$ contribute at most
$e^{-mt}\sum_{j:p_j\geq t}p_j\leq e^{-mt}$; the remaining atoms contribute
at most $\operatorname{Tail}_p(t)$. This proves Eq.~\eqref{eq:sampling-mass}.
\qed

The same upper bound applies to target-proportional sampling without
replacement: conditional on not having selected $j$, its next-draw
probability is at least $p_j$. RIPPLE additionally includes deterministic
high-mass candidates and negatives, so we measure the realized missed target
mass $\delta^T_{i,r}$ rather than treating the bound as an equality for the
full sampler.

\begin{lemma}[Restriction to a candidate set]
\label{lem:restrict}
Let $p$ and $q$ be distributions on a common finite set. If
$p(C^c)=\delta_p$ and $q(C^c)=\delta_q$, then
\begin{equation}
\mathrm{TV}(p,q)
\leq \mathrm{TV}(p|_C,q|_C)+\delta_p+\delta_q.
\end{equation}
\end{lemma}

\paragraph{Proof.}
Extending the conditional distributions by zero outside $C$ gives
$\mathrm{TV}(p,p|_C)=\delta_p$ and
$\mathrm{TV}(q,q|_C)=\delta_q$. The claim follows from the triangle
inequality. \qed

\paragraph{Proof of Corollary~\ref{cor:sampled-topology}.}
Pinsker's inequality and Lemma~\ref{lem:restrict} imply
\begin{equation}
\mathrm{TV}(g^T_{i,r},g^S_{i,r})
\leq
\sqrt{\kappa_{i,r}/2}+\delta^T_{i,r}+\delta^S_{i,r}
=\eta_{i,r}.
\end{equation}
Applying Eq.~\eqref{eq:reverse-tv} to anchors $i$ and $j$ gives
Eq.~\eqref{eq:sampled-topology-bound}. \qed

\subsection{Proof of Proposition~\ref{prop:row-consistency}}
\label{app:proof-gauge}

Let the coincident teacher and student softmax rows on $\Omega_i$ be $p_i$.
For any $j,j'\in\Omega_i$, equality of log-ratios gives
\begin{equation}
\frac{c^S_{ij}-c^S_{ij'}}{\tau_i}
=\log\frac{p_i(j)}{p_i(j')}
=\frac{c^T_{ij}-c^T_{ij'}}{\tau_i}.
\end{equation}
Thus $c^S_{ij}-c^T_{ij}$ is constant in $j$ on $\Omega_i$; denote it by
$a_i$. Conversely, adding a constant to every logit in a row leaves its
softmax unchanged.

For an edge $\{u,v\}$ of the reciprocal-overlap graph,
Eq.~\eqref{eq:row-gauge} gives
$c^S_{uv}=c^T_{uv}+a_u$ and $c^S_{vu}=c^T_{vu}+a_v$.
Teacher and student cosine scores are symmetric, so $a_u=a_v$. Equality
propagates along paths, making the offset constant on each connected
component. \qed
